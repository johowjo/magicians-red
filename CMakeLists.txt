cmake_minimum_required(VERSION 3.20)
project(adaptive_gpu_pta LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(pta_common src/common/parser.cpp src/cpu/reference.cpp)
target_include_directories(pta_common PUBLIC include)
target_compile_options(pta_common PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pta_ref src/main.cpp)
target_link_libraries(pta_ref PRIVATE pta_common)
add_library(pta_test_generator benchmarks/generators/random_workload.cpp)
target_include_directories(pta_test_generator PUBLIC benchmarks/generators)
target_link_libraries(pta_test_generator PUBLIC pta_common)
add_executable(pta_gen benchmarks/generators/main.cpp)
target_link_libraries(pta_gen PRIVATE pta_test_generator)
enable_testing()
add_executable(pta_unit tests/unit/reference_test.cpp)
target_link_libraries(pta_unit PRIVATE pta_common)
add_test(NAME pta_unit COMMAND pta_unit)
add_executable(pta_differential benchmarks/generators/differential_test.cpp)
target_link_libraries(pta_differential PRIVATE pta_test_generator)
add_test(NAME pta_differential COMMAND pta_differential)
